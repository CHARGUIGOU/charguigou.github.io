{%- comment -%} URL de l'image d'un personnage, lue dans _data/characters.yml. Paramètres : key = id du personnage (ex: elsa) — ou directement une URL complète (https://...) ; "-" = pas d'image name = nom original affiché (ex: Ralph) : sert seulement quand key est vide, l'id est alors déduit du nom (minuscules, espaces en tirets : "Mr. Litwak" -> mr-litwak) film = titre du film (facultatif) : si le personnage a une image pour CE film, c'est elle qui est prise px = taille de la miniature en pixels (défaut 160 ; carrée, recadrée depuis le haut) Renvoie seulement l'URL (vide si le personnage ou son image n'existe pas). {%- endcomment -%} {%- assign ci_url = "" -%} {%- assign ci_key = include.key | default: "" | strip -%} {%- if ci_key == "" -%} {%- assign ci_n = include.name | default: "" | split: "<" | first | default: "" | split: "(" | first | default: "" | strip | downcase -%} {%- assign ci_key = ci_n | replace: "’", "" | replace: "'", "" | replace: ".", "" | replace: ",", "" | replace: ":", "" | replace: "&", "" | replace: "&", "" | replace: "#", "" | replace: "!", "" | replace: "?", "" | replace: " ", "-" | replace: "--", "-" -%} {%- endif -%} {%- if ci_key == "-" -%}{%- assign ci_key = "" -%}{%- endif -%} {%- assign ci_k4 = ci_key | slice: 0, 4 -%} {%- if ci_k4 == "http" -%} {%- assign ci_url = ci_key -%} {%- elsif ci_key != "" -%} {%- assign ci_char = site.data.characters.list | where: "id", ci_key | first -%} {%- if ci_char -%} {%- assign ci_url = ci_char.img | default: "" -%} {%- assign ci_film = include.film | default: "" | downcase | strip -%} {%- if ci_film != "" -%} {%- for ci_ap in ci_char.appearances -%} {%- assign ci_apf = ci_ap.film | downcase | strip -%} {%- if ci_apf == ci_film and ci_ap.img and ci_ap.img != "" -%}{%- assign ci_url = ci_ap.img -%}{%- break -%}{%- endif -%} {%- endfor -%} {%- endif -%} {%- endif -%} {%- endif -%} {%- if ci_url != "" -%} {%- assign ci_u4 = ci_url | slice: 0, 4 -%} {%- if ci_u4 != "http" -%}{%- assign ci_url = site.data.characters.image_base | append: ci_url -%}{%- endif -%} {%- assign ci_size = include.px | default: 160 -%} {%- assign ci_repl = "c_fill,w_" | append: ci_size | append: ",h_" | append: ci_size | append: ",g_north" -%} {%- assign ci_url = ci_url | replace: "c_limit,w_500", ci_repl -%} {{- ci_url -}} {%- endif -%}
